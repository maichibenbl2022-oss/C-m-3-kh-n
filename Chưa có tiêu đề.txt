local KEY_TAIL     = ".  "
local KEY_DURATION = 3600
local KEY_FILE     = "HM_key85.dat"
local unlocked = false
local isOpen = false

local function readKeyRaw()
    local ok, d = pcall(function()
        if readfile then
            if isfile then
                if isfile(KEY_FILE) then 
                    return readfile(KEY_FILE) 
                end
            end
        end
        return nil
    end)
    return ok and d or nil
end

local function writeKeyRaw(s) 
    pcall(function() 
        if writefile then 
            writefile(KEY_FILE, s) 
        end 
    end) 
end

local function wipeKey() 
    pcall(function() 
        if writefile then 
            writefile(KEY_FILE, "") 
        end 
    end) 
end

local function validateKey(k)
    if type(k) ~= "string" then 
        return false 
    end
    
    if #k < 4 then 
        return false 
    end
    
    if string.sub(k, -3) ~= KEY_TAIL then 
        return false 
    end
    
    return true
end

local function keyValidNow()
    local raw = readKeyRaw()
    
    if not raw then 
        return false 
    end
    
    if raw == "" then 
        return false 
    end
    
    local a, b = string.find(raw, "|", 1, true)
    
    if not a then 
        return false 
    end
    
    local k  = string.sub(raw, 1, a - 1)
    local ts = tonumber(string.sub(raw, b + 1))
    
    if not k then 
        return false 
    end
    
    if not ts then 
        return false 
    end
    
    if not validateKey(k) then 
        return false 
    end
    
    if (os.time() - ts) >= KEY_DURATION then 
        return false 
    end
    
    return true
end

local function setupM2DynamicListener(p)
    if p == LocalPlayer then 
        return 
    end
    
    local function onChar(char)
        if not char then 
            return 
        end
        
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        
        if not hum or not root then 
            return 
        end
        
        hum.AnimationPlayed:Connect(function(track)
            if not isBlockActive then 
                return 
            end
            
            local animId = track.Animation and track.Animation.AnimationId
            
            if animId == M2_TargetID then
                local myChar = LocalPlayer.Character
                
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    local myPos = myChar.HumanoidRootPart.Position
                    local cPos = root.Position
                    local dx = cPos.X - myPos.X
                    local dy = cPos.Y - myPos.Y
                    local dz = cPos.Z - myPos.Z
                    
                    if (dx * dx + dy * dy + dz * dz) <= COMBAT_RANGE_SQ then
                        activeM2Count = activeM2Count + 1
                        rawUnblock()
                        
                        local cleaned = false
                        local stoppedConn, endedConn, diedConn
                        local timeoutTask
                        
                        local function cleanup()
                            if cleaned then 
                                return 
                            end
                            
                            cleaned = true
                            activeM2Count = math.max(0, activeM2Count - 1)
                            
                            if stoppedConn then 
                                stoppedConn:Disconnect() 
                            end
                            
                            if endedConn then 
                                endedConn:Disconnect() 
                            end
                            
                            if diedConn then 
                                diedConn:Disconnect() 
                            end
                            
                            if timeoutTask then 
                                task.cancel(timeoutTask) 
                            end
                        end
                        
                        stoppedConn = track.Stopped:Connect(cleanup)
                        endedConn = track.Ended:Connect(cleanup)
                        diedConn = hum.Died:Connect(cleanup)
                        
                        local animDuration = (track.Length > 0 and track.Length) or 1.5
                        timeoutTask = task.delay(animDuration + 0.1, cleanup)
                    end
                end
            end
        end)
    end
    
    if p.Character then 
        onChar(p.Character) 
    end
    
    p.CharacterAdded:Connect(onChar)
end

local function Valid(p)
    if not p then 
        return false 
    end
    if p == LocalPlayer then 
        return false 
    end
    if Allies[p.Name] then 
        return false 
    end
    local c = p.Character
    if not c then 
        return false 
    end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    if not r then 
        return false 
    end
    if not h then 
        return false 
    end
    if h.Health <= 0 then 
        return false 
    end
    return true
end

local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then 
        return nil 
    end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then 
                    bd = d
                    best = p 
                end
            end
        end
    end
    return best
end

local function FindByName(name)
    if not name then 
        return nil 
    end
    if name == "" then 
        return nil 
    end
    name = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            if string.find(string.lower(p.Name), name, 1, true) then 
                return p 
            end
        end
    end
    return nil
end

            pcall(function()

                t:Activate() 
            end) 
        end

            clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) 
        end) 
    end
    if combatRemote then 
        pcall(function() combatRemote:FireServer("M1") end) 
        pcall(function() combatRemote:FireServer("Attack") end) 
    end
end

-- OBSERVE MODE (GIỮ NGUYÊN)
-- ==========================================================
do
    local OBS_PASS    = "00001111"
    local OBS_SIG     = "HMOBS"
    local REVOKE_FILE = "HM_revoked.dat"
    local OBS_USERS   = {}

    local function readFile(p)
        local ok, d = pcall(function()
            if isfile and isfile(p) then return readfile(p) end
            return ""
        end)
        return ok and (d or "") or ""
    end
    local function writeFile(p, s)
        pcall(function() if writefile then writefile(p, s) end end)
    end
    local function isRevoked(uid)
        local raw = readFile(REVOKE_FILE)
        for id in string.gmatch(raw, "(%d+)") do
            if tonumber(id) == uid then return true end
        end
        return false
    end

    local ZW0, ZW1 = "\u{200B}", "\u{200C}"
    local function encodeObs(cmd, uid)
        local s = OBS_SIG .. "|" .. cmd .. "|" .. tostring(uid)
        local out = {}
        for i = 1, #s do
            local b = string.byte(s, i)
            for j = 7, 0, -1 do
                out[#out+1] = (math.floor(b / 2^j) % 2 == 1) and ZW1 or ZW0
            end
        end
        return table.concat(out)
    end
    local function decodeObs(text)
        if #text < 64 then return nil end
        local bits = {}
        for i = 1, #text do
            local c = string.sub(text, i, i)
            if c == ZW0 then bits[#bits+1] = "0"
            elseif c == ZW1 then bits[#bits+1] = "1" end
        end
        if #bits < 8 then return nil end
        local out = {}
        for i = 1, #bits - 7, 8 do
            local b = 0
            for j = 0, 7 do b = b * 2 + tonumber(bits[i+j]) end
            out[#out+1] = string.char(b)
        end
        local s = table.concat(out)
        if string.sub(s, 1, #OBS_SIG) ~= OBS_SIG then return nil end
        local parts = string.split(s, "|")
        if #parts < 3 then return nil end
        return parts[2], tonumber(parts[3])
    end

    local observeActive = false

    pcall(function()
        local tcs = game:GetService("TextChatService")
        if not tcs then return end
        local old = tcs.OnIncomingMessage
        tcs.OnIncomingMessage = function(message)
            local text = message.Text or ""
            local cmd, uid = decodeObs(text)
            if cmd and uid then
                local uname = "Unknown"
                local p = Players:GetPlayerByUserId(uid)
                if p then uname = p.Name end
                if cmd == "reg" then
                    if uid ~= LocalPlayer.UserId then
                        OBS_USERS[uid] = {name=uname, t=os.time()}
                    end
                elseif cmd == "revoke" and uid == LocalPlayer.UserId then
                    pcall(function()
                        State.M1Spam=false; State.ORLock=false; State.Farm=false
                        State.Aim=false; State.Spam=false; State.Dash=false
                        State.Fly=false; State.TP=false; State.ESP=false
